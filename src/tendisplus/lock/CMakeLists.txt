add_subdirectory(mgl)

add_library(lock lock.cpp)
target_link_libraries(lock mgl)

if(TENDIS_WITH_UNIT_TESTS)
add_executable(lock_test lock_test.cpp)
if(APPLE)
    # macOS: use -force_load (ld64 linker)
    target_link_libraries(lock_test lock session session_ctx server gtest_main ${SYS_LIBS})
    target_link_libraries(lock_test -Wl,-force_load,$<TARGET_FILE:commands>)
elseif(MSVC)
    # Windows: use /WHOLEARCHIVE
    target_link_libraries(lock_test commands lock session session_ctx server gtest_main ${SYS_LIBS})
    set_target_properties(lock_test PROPERTIES LINK_FLAGS "/WHOLEARCHIVE:commands")
else()
    # Linux/Unix (GCC or Clang): use --whole-archive (GNU ld)
    target_link_libraries(lock_test -Wl,--whole-archive commands -Wl,--no-whole-archive)
    target_link_libraries(lock_test lock session session_ctx server gtest_main ${SYS_LIBS})
endif()
endif()
